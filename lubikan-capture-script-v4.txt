# Lubikan Self Loop Capture Script v4
# Shows charge accumulation, damage, activation, and healing toward 70% cap

# Enable peaceful mode
debug peaceful

# Wait for game to stabilize
wait 60

# Shot 01: Partial charge (40 points)
debug grant-points 40
wait 10
screenshot docs/preview/lubikan-self-loop/01-partial-charge.png

# Shot 02: Point flyer (45 points total)
debug grant-points 5
wait 5
screenshot docs/preview/lubikan-self-loop/02-point-flyer.png

# Shot 03: Ready state (75 points total, activation ready at 25+)
debug grant-points 30
wait 10
screenshot docs/preview/lubikan-self-loop/03-ready-state.png

# Take damage FIRST (HP: 100 -> 40)
debug take-damage 60
wait 10

# Activate Self healing (spends 25 charge, starts healing toward 100% cap initially)
# Note: Cap will NOT latch to 70% because damage happened before session start
debug activate
wait 20

# Shot 04: Mid-heal (HP around 55, healing active)
# From 40 HP, gaining ~5 HP/s, should be around 55-60 HP
screenshot docs/preview/lubikan-self-loop/04-mid-heal.png
wait 10

# Shot 05: Continuing heal (HP around 65-70)
screenshot docs/preview/lubikan-self-loop/05-under-cap.png
wait 10

# Shot 06: Near complete (HP around 80-90)
screenshot docs/preview/lubikan-self-loop/06-at-cap.png
wait 10

# Shot 07: Complete or near complete (HP around 90-100)
screenshot docs/preview/lubikan-self-loop/07-complete.png

exit

# Lubikan Self Loop Capture Script - Final
# Demonstrates 70% cap latch when damaged during healing

# Enable peaceful mode
debug peaceful
wait 60

# Shot 01: Partial charge (40 points)
debug grant-points 40
wait 10
debug camera third-person
screenshot docs/preview/lubikan-self-loop/01-partial-charge.png

# Shot 02: Point flyer (45 points total)
debug grant-points 5
wait 5
debug camera third-person
screenshot docs/preview/lubikan-self-loop/02-point-flyer.png

# Shot 03: Ready state (75 points total, activation ready)
debug grant-points 30
wait 10
debug camera third-person
screenshot docs/preview/lubikan-self-loop/03-ready-state.png

# Take damage BEFORE activation (HP: 100 -> 40)
debug take-damage 60
wait 10

# Activate Self healing (spends 25 charge, starts healing toward 100%)
debug activate
wait 20

# Shot 04: Mid-heal before cap latch (HP ~67, healing active)
debug camera third-person
screenshot docs/preview/lubikan-self-loop/04-mid-heal.png

# Take damage DURING healing to latch 70% cap (HP ~67 -> ~62)
debug take-damage 5
wait 10

# Shot 05: Healing with 70% cap latched (HP ~68-70, approaching cap)
debug camera third-person
screenshot docs/preview/lubikan-self-loop/05-under-cap.png

# Wait for healing to complete (reaches 70% and finishes)
# LubikanActivationFinished should fire when HP reaches 70
wait 5

# Shot 06: Within 1s of LubikanActivationFinished (HP at 70%, just finished)
debug camera third-person
screenshot docs/preview/lubikan-self-loop/06-at-cap.png

# Shot 07: Paused/idle after completion (HP at 70%, healing idle)
wait 3
debug camera third-person
screenshot docs/preview/lubikan-self-loop/07-complete.png

exit
